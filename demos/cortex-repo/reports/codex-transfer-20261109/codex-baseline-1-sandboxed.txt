"""Read schema/models.yaml and render the derived files."""

import hashlib
from pathlib import Path

import yaml

ROOT = Path(__file__).resolve().parents[1]
SCHEMA = ROOT / "schema" / "models.yaml"
TYPES = {"int": "int", "str": "str", "bool": "bool", "float": "float"}
SQL = {"int": "INTEGER", "str": "TEXT", "bool": "INTEGER", "float": "REAL"}


def load():
    models = yaml.safe_load(SCHEMA.read_text())["models"]
    for name, model in models.items():
        for field, spec in model["fields"].items():
            if spec.get("type") not in TYPES:
                raise SystemExit(f"schema: {name}.{field} has unsupported type {spec.get('type')!r}")
    return models


def digest():
    return hashlib.sha256(SCHEMA.read_bytes()).hexdigest()[:12]


def python_default(spec):
    if "default" not in spec:
        return "None" if not spec.get("primary") else None
    return repr(spec["default"])


def render_models(models):
    lines = [f"# GENERATED by `make generate` from schema/models.yaml (schema {digest()}). Do not edit.", ""]
    lines += ["from dataclasses import dataclass", "", f'SCHEMA_DIGEST = "{digest()}"', ""]
    for name, model in models.items():
        lines += ["", "@dataclass", f"class {name}:"]
        required = [(f, s) for f, s in model["fields"].items() if s.get("primary")]
        optional = [(f, s) for f, s in model["fields"].items() if not s.get("primary")]
        for field, spec in required:
            lines.append(f"    {field}: {TYPES[spec['type']]}")
        for field, spec in optional:
            lines.append(f"    {field}: {TYPES[spec['type']]} | None = {python_default(spec)}")
        lines.append("")
    lines += ["", "TABLES = {" + ", ".join(f'"{m["table"]}": {n}' for n, m in models.items()) + "}", ""]
    return "\n".join(lines)

"""SQLite persistence for the generated models."""

import sqlite3
from dataclasses import asdict, fields

from shopkit.models import TABLES

MODELS = {model: table for table, model in TABLES.items()}


def connect(path):
    connection = sqlite3.connect(path)
    connection.row_factory = sqlite3.Row
    return connection


def insert(connection, record):
    values = {key: value for key, value in asdict(record).items() if value is not None}
    columns = ", ".join(values)
    marks = ", ".join("?" for _ in values)
    cursor = connection.execute(f"INSERT INTO {MODELS[type(record)]} ({columns}) VALUES ({marks})", list(values.values()))
    connection.commit()
    return cursor.lastrowid


def get(connection, model, record_id):
    row = connection.execute(f"SELECT * FROM {MODELS[model]} WHERE id = ?", (record_id,)).fetchone()
    if row is None:
        return None
    return model(**{field.name: row[field.name] for field in fields(model)})


def update(connection, record):
    values = {key: value for key, value in asdict(record).items() if key != "id"}
    assignments = ", ".join(f"{key} = ?" for key in values)
    connection.execute(f"UPDATE {MODELS[type(record)]} SET {assignments} WHERE id = ?", [*values.values(), record.id])
    connection.commit()

"""Database migrations: apply them to a SQLite file, or create one for a schema change.

python tools/migrate.py apply DB
python tools/migrate.py new NAME
"""

import re
import sqlite3
import sys

import schema

MIGRATIONS = schema.ROOT / "migrations"


def files():
    return sorted(MIGRATIONS.glob("[0-9][0-9][0-9][0-9]_*.sql"))


def apply(db):
    connection = sqlite3.connect(db)
    for path in files():
        connection.executescript(path.read_text())
    connection.commit()
    return connection


def database_columns(connection):
    tables = [row[0] for row in connection.execute("SELECT name FROM sqlite_master WHERE type = 'table'")]
    return {table: {row[1]: row[2] for row in connection.execute(f"PRAGMA table_info({table})")} for table in tables}


def new(name):
    if not re.fullmatch(r"[a-z0-9_]+", name):
        raise SystemExit("migration NAME must be lowercase letters, digits and underscores")
    current = database_columns(apply(":memory:"))
    statements = []
    for table, wanted in schema.columns(schema.load()).items():
        have = current.get(table)
        if have is None:
            body = ", ".join(f"{column} {kind}" + (" PRIMARY KEY" if column == "id" else "") for column, kind in wanted.items())
            statements.append(f"CREATE TABLE {table} ({body});")
            continue
        statements += [f"ALTER TABLE {table} ADD COLUMN {column} {kind};" for column, kind in wanted.items() if column not in have]
    if not statements:
        raise SystemExit("migration: the database already matches schema/models.yaml; nothing to do")
    number = int(files()[-1].name[:4]) + 1 if files() else 1
    path = MIGRATIONS / f"{number:04d}_{name}.sql"
    path.write_text("\n".join(statements) + "\n")
    print(f"created {path.relative_to(schema.ROOT)}")


if __name__ == "__main__":
    if len(sys.argv) == 3 and sys.argv[1] == "apply":
        apply(sys.argv[2]).close()
    elif len(sys.argv) == 3 and sys.argv[1] == "new":
        new(sys.argv[2])
    else:
        raise SystemExit(__doc__)
