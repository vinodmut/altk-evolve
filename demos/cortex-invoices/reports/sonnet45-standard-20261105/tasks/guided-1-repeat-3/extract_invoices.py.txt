# Write items.csv
with open('items.csv', 'w', newline='') as f:
    writer = csv.writer(f)
    writer.writerow(['invoice_id', 'vendor', 'description', 'quantity', 'unit_price', 'line_amount'])
    
    for invoice_id, data in sorted(invoices.items()):
        vendor = data['vendor']
        for item in data['items']:
            writer.writerow([
                invoice_id,
                vendor,
                item['description'],
                item['qty'],
                item['unit_price'],
                item['amount']
            ])

# Write reconciliation.csv
with open('reconciliation.csv', 'w', newline='') as f:
    writer = csv.writer(f)
    writer.writerow(['invoice_id', 'subtotal', 'tax', 'printed_total', 'computed_total', 'difference', 'status'])
    
    for invoice_id, data in sorted(invoices.items()):
        # Calculate computed subtotal from line items
        computed_subtotal = sum(float(item['amount']) for item in data['items'])
        
        # Get printed values
        printed_subtotal = float(data['subtotal'])
        printed_tax = float(data['tax'])
        printed_total = float(data['total'])
        
        # Compute total
        computed_total = computed_subtotal + printed_tax
        
        # Calculate difference (computed - printed)
        difference = computed_total - printed_total
        
        # Determine status
        status = 'matched' if abs(difference) < 0.01 else 'discrepancy'
        
        writer.writerow([
            invoice_id,
            f'{computed_subtotal:.2f}',
            f'{printed_tax:.2f}',
            f'{printed_total:.2f}',
            f'{computed_total:.2f}',
            f'{difference:.2f}',
            status
        ])
