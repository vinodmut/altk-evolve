# Write items.csv
with open('items.csv', 'w', newline='') as f:
    writer = csv.writer(f)
    writer.writerow(['invoice_id', 'vendor', 'description', 'quantity', 'unit_price', 'line_amount'])
    
    for invoice_id in sorted(invoices.keys()):
        vendor = invoices[invoice_id]['vendor']
        for item in invoices[invoice_id]['items']:
            writer.writerow([
                invoice_id,
                vendor,
                item['description'],
                item['quantity'],
                item['unit_price'],
                item['line_amount']
            ])

# Calculate reconciliation
reconciliation = []

for invoice_id in sorted(invoices.keys()):
    invoice = invoices[invoice_id]
    
    # Calculate subtotal from line items
    subtotal = Decimal('0')
    for item in invoice['items']:
        subtotal += Decimal(item['line_amount'])
    
    tax = Decimal(invoice['tax'])
    printed_total = Decimal(invoice['printed_total'])
    computed_total = subtotal + tax
    difference = computed_total - printed_total
    
    status = 'matched' if difference == 0 else 'discrepancy'
    
    reconciliation.append({
        'invoice_id': invoice_id,
        'subtotal': str(subtotal),
        'tax': str(tax),
        'printed_total': str(printed_total),
        'computed_total': str(computed_total),
        'difference': str(difference),
        'status': status
    })

# Write reconciliation.csv
with open('reconciliation.csv', 'w', newline='') as f:
    writer = csv.writer(f)
    writer.writerow(['invoice_id', 'subtotal', 'tax', 'printed_total', 'computed_total', 'difference', 'status'])
    
    for rec in reconciliation:
        writer.writerow([
            rec['invoice_id'],
            rec['subtotal'],
            rec['tax'],
            rec['printed_total'],
            rec['computed_total'],
            rec['difference'],
            rec['status']
        ])

print("Extraction complete!")
print("\nReconciliation Summary:")
for rec in reconciliation:
    print(f"Invoice {rec['invoice_id']}: {rec['status']}")
    if rec['status'] == 'discrepancy':
        print(f"  Difference: {rec['difference']}")
