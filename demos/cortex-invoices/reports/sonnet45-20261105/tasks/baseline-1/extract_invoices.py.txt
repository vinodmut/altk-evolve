# Write items.csv
with open('items.csv', 'w', newline='') as f:
    writer = csv.writer(f)
    writer.writerow(['invoice_id', 'vendor', 'description', 'quantity', 'unit_price', 'line_amount'])
    
    for invoice_id in sorted(invoices.keys()):
        invoice = invoices[invoice_id]
        for item in invoice['items']:
            description, qty, unit_price, line_amount = item
            writer.writerow([invoice_id, invoice['vendor'], description, qty, unit_price, line_amount])

# Write reconciliation.csv
with open('reconciliation.csv', 'w', newline='') as f:
    writer = csv.writer(f)
    writer.writerow(['invoice_id', 'subtotal', 'tax', 'printed_total', 'computed_total', 'difference', 'status'])
    
    for invoice_id in sorted(invoices.keys()):
        invoice = invoices[invoice_id]
        
        # Compute subtotal from line items
        computed_subtotal = Decimal('0')
        for item in invoice['items']:
            line_amount = Decimal(item[3])
            computed_subtotal += line_amount
        
        subtotal = Decimal(invoice['subtotal'])
        tax = Decimal(invoice['tax'])
        printed_total = Decimal(invoice['total'])
        computed_total = computed_subtotal + tax
        difference = computed_total - printed_total
        
        status = 'matched' if difference == 0 else 'discrepancy'
        
        writer.writerow([
            invoice_id,
            f"{subtotal:.2f}",
            f"{tax:.2f}",
            f"{printed_total:.2f}",
            f"{computed_total:.2f}",
            f"{difference:.2f}",
            status
        ])
