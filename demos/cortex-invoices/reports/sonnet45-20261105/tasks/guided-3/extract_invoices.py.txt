# Write items.csv
with open('items.csv', 'w', newline='') as f:
    writer = csv.writer(f)
    writer.writerow(['invoice_id', 'vendor', 'description', 'quantity', 'unit_price', 'line_amount'])
    
    for invoice_id, invoice in invoices.items():
        for item in invoice['line_items']:
            writer.writerow([
                invoice_id,
                invoice['vendor'],
                item['description'],
                item['quantity'],
                item['unit_price'],
                item['line_amount']
            ])

print("Created items.csv with", sum(len(inv['line_items']) for inv in invoices.values()), "line items")

# Write reconciliation.csv
with open('reconciliation.csv', 'w', newline='') as f:
    writer = csv.writer(f)
    writer.writerow(['invoice_id', 'subtotal', 'tax', 'printed_total', 'computed_total', 'difference', 'status'])
    
    for invoice_id, invoice in invoices.items():
        subtotal = Decimal(invoice['subtotal'])
        tax = Decimal(invoice['tax'])
        printed_total = Decimal(invoice['total'])
        
        # Compute total from line amounts
        computed_subtotal = sum(Decimal(item['line_amount']) for item in invoice['line_items'])
        computed_total = computed_subtotal + tax
        
        # Calculate difference
        difference = computed_total - printed_total
        
        # Determine status
        status = 'matched' if difference == 0 else 'discrepancy'
        
        writer.writerow([
            invoice_id,
            f"{subtotal:.2f}",
            f"{tax:.2f}",
            f"{printed_total:.2f}",
            f"{computed_total:.2f}",
            f"{difference:.2f}",
            status
        ])
        
        # Check if subtotal matches computed subtotal
        if computed_subtotal != subtotal:
            print(f"WARNING: Invoice {invoice_id} - Computed subtotal {computed_subtotal:.2f} != Printed subtotal {subtotal:.2f}")
        
        if difference != 0:
            print(f"DISCREPANCY: Invoice {invoice_id} - Difference: {difference:.2f}")

print("Created reconciliation.csv")
