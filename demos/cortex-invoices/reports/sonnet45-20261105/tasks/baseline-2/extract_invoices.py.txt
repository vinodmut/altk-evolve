# Write items.csv
with open('items.csv', 'w', newline='') as f:
    writer = csv.writer(f)
    writer.writerow(['invoice_id', 'vendor', 'description', 'quantity', 'unit_price', 'line_amount'])
    
    for invoice_id, data in sorted(invoices.items()):
        for item in data['items']:
            writer.writerow([
                invoice_id,
                data['vendor'],
                item['description'],
                item['qty'],
                item['unit_price'],
                item['amount']
            ])

print(f"Created items.csv with {sum(len(data['items']) for data in invoices.values())} line items")

# Perform reconciliation
reconciliation = []
for invoice_id, data in sorted(invoices.items()):
    # Calculate sum of line amounts
    computed_subtotal = sum(Decimal(item['amount']) for item in data['items'])
    
    # Get printed values
    printed_subtotal = Decimal(data['subtotal'])
    tax = Decimal(data['tax'])
    printed_total = Decimal(data['total'])
    
    # Compute total
    computed_total = computed_subtotal + tax
    
    # Calculate difference (computed - printed)
    difference = computed_total - printed_total
    
    # Determine status
    status = 'matched' if difference == 0 else 'discrepancy'
    
    reconciliation.append({
        'invoice_id': invoice_id,
        'subtotal': str(computed_subtotal),
        'tax': str(tax),
        'printed_total': str(printed_total),
        'computed_total': str(computed_total),
        'difference': str(difference),
        'status': status
    })

# Write reconciliation.csv
with open('reconciliation.csv', 'w', newline='') as f:
    writer = csv.DictWriter(f, fieldnames=['invoice_id', 'subtotal', 'tax', 'printed_total', 'computed_total', 'difference', 'status'])
    writer.writeheader()
    writer.writerows(reconciliation)

print(f"Created reconciliation.csv with {len(reconciliation)} invoices")
print("\n" + "=" * 80)
print("RECONCILIATION SUMMARY")
print("=" * 80)

for rec in reconciliation:
    print(f"\nInvoice {rec['invoice_id']}: {rec['status'].upper()}")

    print(f"  Tax: ${rec['tax']}")
    print(f"  Printed Total: ${rec['printed_total']}")
    print(f"  Computed Total: ${rec['computed_total']}")
    print(f"  Difference: ${rec['difference']}")
    if rec['status'] == 'discrepancy':
        print(f"  *** DISCREPANCY FOUND ***")

# Count discrepancies
discrepancies = [r for r in reconciliation if r['status'] == 'discrepancy']
if discrepancies:
