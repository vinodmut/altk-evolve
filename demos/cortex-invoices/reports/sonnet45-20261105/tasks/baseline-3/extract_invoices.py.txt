# Write items.csv
with open('items.csv', 'w', newline='') as f:
    writer = csv.writer(f)
    writer.writerow(['invoice_id', 'vendor', 'description', 'quantity', 'unit_price', 'line_amount'])
    
    for invoice_id, data in sorted(invoices.items()):
        vendor = data['vendor']
        for item in data['items']:
            writer.writerow([
                invoice_id,
                vendor,
                item['description'],
                item['qty'],
                item['unit_price'],
                item['line_amount']
            ])

# Write reconciliation.csv
with open('reconciliation.csv', 'w', newline='') as f:
    writer = csv.writer(f)
    writer.writerow(['invoice_id', 'subtotal', 'tax', 'printed_total', 'computed_total', 'difference', 'status'])
    
    for invoice_id, data in sorted(invoices.items()):
        subtotal = Decimal(data['subtotal'])
        tax = Decimal(data['tax'])
        printed_total = Decimal(data['total'])
        computed_total = subtotal + tax
        difference = computed_total - printed_total
        status = 'matched' if difference == 0 else 'discrepancy'
        
        writer.writerow([
            invoice_id,
            data['subtotal'],
            data['tax'],
            data['total'],
            f'{computed_total:.2f}',
            f'{difference:.2f}',
            status
        ])
