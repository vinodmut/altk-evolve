# Write items.csv

# Write reconciliation.csv
with open('reconciliation.csv', 'w', newline='') as f:
    writer = csv.writer(f)
    writer.writerow(['invoice_id', 'subtotal', 'tax', 'printed_total', 'computed_total', 'difference', 'status'])
    
    for invoice_id, invoice_data in invoices.items():
        # Calculate subtotal from line items
        computed_subtotal = sum(Decimal(item['line_amount']) for item in invoice_data['items'])
        
        # Get printed values
        printed_subtotal = Decimal(invoice_data['subtotal'])
        tax = Decimal(invoice_data['tax'])
        printed_total = Decimal(invoice_data['total'])
        
        # Compute total
        computed_total = computed_subtotal + tax
        
        # Calculate difference
        difference = computed_total - printed_total
        
        # Determine status
        status = 'matched' if difference == 0 else 'discrepancy'
        
        writer.writerow([
            invoice_id,
            f'{printed_subtotal:.2f}',
            f'{tax:.2f}',
            f'{printed_total:.2f}',
            f'{computed_total:.2f}',
            f'{difference:.2f}',
            status
        ])
        
        # Report discrepancies
        if status == 'discrepancy':
            print(f"DISCREPANCY FOUND in invoice {invoice_id}:")
            print(f"  Computed subtotal: {computed_subtotal:.2f}")
            print(f"  Printed subtotal: {printed_subtotal:.2f}")
            print(f"  Tax: {tax:.2f}")
            print(f"  Computed total: {computed_total:.2f}")
            print(f"  Printed total: {printed_total:.2f}")
            print(f"  Difference: {difference:.2f}")

print("reconciliation.csv created successfully")
