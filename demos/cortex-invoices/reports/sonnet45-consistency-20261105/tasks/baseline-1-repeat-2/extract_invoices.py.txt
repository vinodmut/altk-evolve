# Write items.csv
with open('items.csv', 'w', newline='') as f:
    writer = csv.writer(f)
    writer.writerow(['invoice_id', 'vendor', 'description', 'quantity', 'unit_price', 'line_amount'])
    
    for invoice_id, data in sorted(invoices.items()):
        for item in data['items']:
            writer.writerow([
                invoice_id,
                data['vendor'],
                item['description'],
                item['qty'],
                item['unit_price'],
                item['amount']
            ])

# Calculate reconciliation
reconciliation = []
for invoice_id, data in sorted(invoices.items()):
    # Calculate subtotal from line items
    subtotal = sum(Decimal(item['amount']) for item in data['items'])
    tax = Decimal(data['tax'])
    printed_total = Decimal(data['total'])
    computed_total = subtotal + tax
    difference = computed_total - printed_total
    status = 'matched' if difference == 0 else 'discrepancy'
    
    reconciliation.append({
        'invoice_id': invoice_id,
        'subtotal': str(subtotal),
        'tax': str(tax),
        'printed_total': str(printed_total),
        'computed_total': str(computed_total),
        'difference': str(difference),
        'status': status
    })

# Write reconciliation.csv
with open('reconciliation.csv', 'w', newline='') as f:
    writer = csv.writer(f)
    writer.writerow(['invoice_id', 'subtotal', 'tax', 'printed_total', 'computed_total', 'difference', 'status'])
    
    for rec in reconciliation:
        writer.writerow([
            rec['invoice_id'],
            rec['subtotal'],
            rec['tax'],
            rec['printed_total'],
            rec['computed_total'],
            rec['difference'],
            rec['status']
        ])

# Print reconciliation results
print("Reconciliation Results:")
