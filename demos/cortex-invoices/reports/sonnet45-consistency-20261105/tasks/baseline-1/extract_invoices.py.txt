# Write items.csv
with open('items.csv', 'w', newline='') as f:
    writer = csv.writer(f)
    writer.writerow(['invoice_id', 'vendor', 'description', 'quantity', 'unit_price', 'line_amount'])
    
    for invoice_id, data in sorted(invoices.items()):
        for item in data['items']:
            description, qty, unit_price, line_amount = item
            writer.writerow([invoice_id, data['vendor'], description, qty, unit_price, line_amount])

# Perform reconciliation
reconciliation = []
for invoice_id, data in sorted(invoices.items()):
    # Calculate subtotal from line items
    subtotal = sum(Decimal(str(item[3])) for item in data['items'])
    tax = Decimal(str(data['tax']))
    printed_total = Decimal(str(data['total']))
    computed_total = subtotal + tax
    difference = computed_total - printed_total
    status = 'matched' if difference == 0 else 'discrepancy'
    
    reconciliation.append({
        'invoice_id': invoice_id,
        'subtotal': float(subtotal),
        'tax': float(tax),
        'printed_total': float(printed_total),
        'computed_total': float(computed_total),
        'difference': float(difference),
        'status': status
    })

# Write reconciliation.csv
with open('reconciliation.csv', 'w', newline='') as f:
    writer = csv.writer(f)
    writer.writerow(['invoice_id', 'subtotal', 'tax', 'printed_total', 'computed_total', 'difference', 'status'])
    
    for rec in reconciliation:
        writer.writerow([
            rec['invoice_id'],
            rec['subtotal'],
            rec['tax'],
            rec['printed_total'],
            rec['computed_total'],
            rec['difference'],
            rec['status']
        ])

print("Extraction complete!")
print("\nReconciliation Summary:")
for rec in reconciliation:
    print(f"Invoice {rec['invoice_id']}: {rec['status'].upper()}")
    if rec['status'] == 'discrepancy':
        print(f"  Difference: ${rec['difference']:.2f}")
