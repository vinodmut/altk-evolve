from decimal import Decimal

items = [
    ('Site survey, area 2', 1, 448.50, 448.50),
    ('Travel time, route 2', 2.25, 64.80, 145.80),
    ('Calibration standard, set 2', 2, 37.25, 74.50),

    ('Cable assemblies, shielded, length 2 m', 5, 11.40, 57.00),
    ('Previously billed visit credit, ticket 2', 1, -26.00, -26.00),
]

print("Invoice 002093 - Line by line verification:")

for desc, qty, unit_price, amount in items:

    total += item_total
