# Invoice totals
invoices = {
    '002071': {'subtotal': Decimal('171.55'), 'tax': Decimal('22.17'), 'total': Decimal('193.72')},
    '002082': {'subtotal': Decimal('2841.18'), 'tax': Decimal('154.28'), 'total': Decimal('2995.46')},
    '002093': {'subtotal': Decimal('775.60'), 'tax': Decimal('90.64'), 'total': Decimal('866.26')},

# Write items.csv
with open('items.csv', 'w', newline='') as f:
    writer = csv.DictWriter(f, fieldnames=['invoice_id', 'vendor', 'description', 'quantity', 'unit_price', 'line_amount'])
    writer.writeheader()
    for item in items:
        writer.writerow(item)

print(f"Wrote {len(items)} items to items.csv")

# Calculate reconciliation
reconciliation = []
for invoice_id in sorted(invoices.keys()):
    invoice_items = [item for item in items if item['invoice_id'] == invoice_id]
    computed_subtotal = sum(Decimal(item['line_amount']) for item in invoice_items)
    tax = invoices[invoice_id]['tax']
    printed_total = invoices[invoice_id]['total']
    computed_total = computed_subtotal + tax
    difference = computed_total - printed_total
    status = 'matched' if difference == 0 else 'discrepancy'
    
    reconciliation.append({
        'invoice_id': invoice_id,
        'subtotal': computed_subtotal,
        'tax': tax,
        'printed_total': printed_total,
        'computed_total': computed_total,
        'difference': difference,
        'status': status
    })
    
    if status == 'discrepancy':
        print(f"WARNING: Discrepancy in {invoice_id}: difference = {difference}")

# Write reconciliation.csv
with open('reconciliation.csv', 'w', newline='') as f:
    writer = csv.DictWriter(f, fieldnames=['invoice_id', 'subtotal', 'tax', 'printed_total', 'computed_total', 'difference', 'status'])
    writer.writeheader()
    for rec in reconciliation:
        writer.writerow(rec)

print(f"Wrote {len(reconciliation)} reconciliation records to reconciliation.csv")

# Summary
discrepancies = [r for r in reconciliation if r['status'] == 'discrepancy']
if discrepancies:
    print(f"\n{len(discrepancies)} DISCREPANCY(IES) FOUND:")
    for d in discrepancies:
        print(f"  Invoice {d['invoice_id']}: difference = {d['difference']}")
else:
    print("\nAll invoices reconciled successfully!")
